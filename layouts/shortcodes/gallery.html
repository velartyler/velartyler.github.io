{{- /* Photo carousel. Usage, inside a post that is a folder (page bundle): {{< gallery >}} Shows every image in the post's "gallery" subfolder, in filename order. Optional per-photo descriptions go in the post's front matter (see README instructions). */ -}} {{- $imgs := .Page.Resources.Match "gallery/*" -}} {{- $imgs = where $imgs "ResourceType" "image" -}} {{- $total := len $imgs -}} {{- if $total }} {{- $id := printf "gallery-%d" .Ordinal }}
{{- range $i, $img := $imgs }} {{- $show := $img }} {{- if and (ne $img.MediaType.SubType "svg") (gt $img.Width 1600) }}{{ $show = $img.Resize "1600x" }}{{ end }} {{- $alt := $img.Params.alt | default (printf "Photo %d of %d" (add $i 1) $total) }} {{- end }}
{{- if gt $total 1 }}
1 / {{ $total }}
{{- end }}
{{- if gt $total 1 }} {{- end }}
{{- end }}
